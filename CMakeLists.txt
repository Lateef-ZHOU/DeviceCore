cmake_minimum_required(VERSION 3.28)
project(LidarDeviceCore LANGUAGES CXX)
include(cmake/Dependencies.cmake)
find_package(Qt6 6.8.3 REQUIRED COMPONENTS Core)
if(NOT TARGET Lidar::HandleCore)
    message(FATAL_ERROR "DeviceCore requires the Lidar::HandleCore target")
endif()
add_library(LidarDeviceCore STATIC include/Device/Base/DeviceBase.h)
add_library(Lidar::DeviceCore ALIAS LidarDeviceCore)
set_target_properties(LidarDeviceCore PROPERTIES AUTOMOC ON)
# 导出 QObject 元数据供消费侧的 QML_FOREIGN 使用；不链接 Qml。
qt_extract_metatypes(LidarDeviceCore)
target_compile_features(LidarDeviceCore PUBLIC cxx_std_20)
target_compile_definitions(LidarDeviceCore PUBLIC QT_NO_KEYWORDS NOMINMAX)
target_include_directories(LidarDeviceCore PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(LidarDeviceCore PUBLIC Lidar::HandleCore Qt6::Core)
if(MSVC)
    target_compile_options(LidarDeviceCore PRIVATE /W4)
endif()

# 作为依赖时不接管宿主的测试开关。
if(PROJECT_IS_TOP_LEVEL)
    include(CTest)
    if(BUILD_TESTING)
        add_subdirectory(tests)
    endif()
endif()
