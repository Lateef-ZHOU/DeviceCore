cmake_minimum_required(VERSION 3.28)
project(LidarDeviceCore LANGUAGES CXX)
include(cmake/Dependencies.cmake)
find_package(Qt6 6.8.3 REQUIRED COMPONENTS Core)
if(NOT TARGET Lidar::HandleCore)
    message(FATAL_ERROR "DeviceCore requires the Lidar::HandleCore target")
endif()
# DeviceBase 不再是 QObject；本库全部为头文件，元对象由 Handle 和具体末端生成。
add_library(LidarDeviceCore INTERFACE)
add_library(Lidar::DeviceCore ALIAS LidarDeviceCore)
target_compile_features(LidarDeviceCore INTERFACE cxx_std_20)
target_compile_definitions(LidarDeviceCore INTERFACE QT_NO_KEYWORDS NOMINMAX)
target_include_directories(LidarDeviceCore INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(LidarDeviceCore INTERFACE Lidar::HandleCore Qt6::Core)

# 作为依赖时不接管宿主的测试开关。
if(PROJECT_IS_TOP_LEVEL)
    include(CTest)
    if(BUILD_TESTING)
        add_subdirectory(tests)
    endif()
endif()
